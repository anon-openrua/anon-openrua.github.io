#!/usr/bin/env python3
"""Save fresh cameras, depth, intrinsics, joints and TF; read-only ROS capture."""
import json,time,datetime,pathlib
import rclpy,tf2_ros,cv2,numpy as np,yaml

for c in m['sensors']:
 if c['kind']!='camera':continue
 for key,typ in [('color',Image),('depth',Image),('info',CameraInfo)]:
  name=c['name']+'_'+key
  subs.append(n.create_subscription(typ,c[key],lambda msg,k=name:data.__setitem__(k,msg),1))
subs.append(n.create_subscription(JointState,'/joint_states',lambda msg:data.__setitem__('joints',msg),1))
t=time.time()

out=pathlib.Path('observations')/datetime.datetime.now(datetime.timezone.utc).strftime('%Y%m%dT%H%M%SZ');out.mkdir(parents=True)
meta={'utc':out.name,'frames':{}}
for k,v in data.items():
 if k.endswith('_color'):cv2.imwrite(str(out/(k+'.png')),bridge.imgmsg_to_cv2(v,'bgr8'))
 elif k.endswith('_depth'):np.save(str(out/(k+'.npy')),bridge.imgmsg_to_cv2(v,'passthrough'))
 elif k.endswith('_info'):meta[k]={'k':list(v.k),'frame':v.header.frame_id}
 else:meta[k]=dict(zip(v.name,v.position))
for frame in ['panda_link8','robotiq_pinch','zed_wrist_left_camera_optical_frame']:
 try:
  x=b.lookup_transform('panda_link0',frame,Time()).transform
  meta['frames'][frame]={'p':[x.translation.x,x.translation.y,x.translation.z],'q':[x.rotation.x,x.rotation.y,x.rotation.z,x.rotation.w]}
 except Exception as e:meta['frames'][frame]={'error':str(e)}
