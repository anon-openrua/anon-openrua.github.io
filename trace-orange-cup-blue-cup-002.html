#!/usr/bin/env python3
"""Save fresh RGB/depth, intrinsics, robot joints and TF in a UTC directory."""
import json,time
from datetime import datetime,timezone
from pathlib import Path
import cv2,numpy as np,rclpy,yaml,tf2_ros
from cv_bridge import CvBridge

 if c['kind']!='camera': continue

subs.append(n.create_subscription(JointState,'/joint_states',lambda msg:got.setdefault('joints',msg),1))
end=time.monotonic()+10

data={}; bridge=CvBridge()
for tag,msg in got.items():
 if isinstance(msg,Image):
  a=bridge.imgmsg_to_cv2(msg,'bgr8' if tag.endswith('color') else 'passthrough')
  if tag.endswith('color'): cv2.imwrite(str(p/(tag+'.png')),a)
  else: np.save(p/(tag+'.npy'),a)
  data[tag]={'frame':msg.header.frame_id,'stamp':msg.header.stamp.sec+1e-9*msg.header.stamp.nanosec,'encoding':msg.encoding}
 elif isinstance(msg,CameraInfo): data[tag]={'frame':msg.header.frame_id,'k':list(msg.k)}
 else: data['joints']=dict(zip(msg.name,msg.position))
for frame in ['panda_link8','robotiq_pinch','zed_wrist_left_camera_optical_frame']:
 try:
  t=b.lookup_transform('panda_link0',frame,Time()).transform
  data[frame]={'xyz':[t.translation.x,t.translation.y,t.translation.z],'quat':[t.rotation.x,t.rotation.y,t.rotation.z,t.rotation.w]}
 except Exception as e: data[frame]={'error':str(e)}
(p/'state.json').write_text(json.dumps(data,indent=2)); print(p); print(json.dumps({k:v for k,v in data.items() if k in ['panda_link8','robotiq_pinch','zed_wrist_left_camera_optical_frame','joints']},indent=2))
