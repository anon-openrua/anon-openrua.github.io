#!/usr/bin/env python3
"""Save timestamped RGB/depth, intrinsics, joints and available TF. No motion."""
from pathlib import Path
from datetime import datetime, timezone
import json, time
import rclpy, yaml, cv2, numpy as np
from sensor_msgs.msg import Image, CameraInfo, JointState
from cv_bridge import CvBridge

    return lambda msg: data.__setitem__(key,msg)

meta={}; bridge=CvBridge()
for key,msg in data.items():
    if key.endswith('color'): cv2.imwrite(str(out/(key+'.png')),bridge.imgmsg_to_cv2(msg,'bgr8'))
    elif key.endswith('depth'): np.save(out/(key+'.npy'),bridge.imgmsg_to_cv2(msg,'passthrough'))
    elif key.endswith('info'): meta[key]={'k':list(msg.k),'frame':msg.header.frame_id}
    else: meta[key]=dict(zip(msg.name,msg.position))
    if hasattr(msg,'header'): meta[key+'_stamp']=[msg.header.stamp.sec,msg.header.stamp.nanosec]

    except Exception as e: meta[f]=str(e)
