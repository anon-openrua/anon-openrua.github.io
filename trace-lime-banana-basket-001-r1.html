#!/usr/bin/env python3
"""Capture timestamped camera, depth, joint, and TF evidence. Requires ROS packages, cv_bridge, numpy, scipy, PyYAML."""
import json,time,datetime,pathlib
import numpy as np

from tf2_ros import Buffer,TransformListener
m=yaml.safe_load(open('/robot/machine.yaml'))
rclpy.init(); n=rclpy.create_node('task_observe'); b=Buffer(); listener=TransformListener(b,n)
got={}; subs=[]
def add(key,topic,typ):
    subs.append(n.create_subscription(typ,topic,lambda msg:got.__setitem__(key,msg),1))
add('joints','/joint_states',JointState)
for c in m['sensors']:
    if c['kind']=='camera':
        for kind,typ in [('color',Image),('depth',Image),('info',CameraInfo)]: add(c['name']+'_'+kind,c[kind],typ)
end=time.monotonic()+12
while time.monotonic()<end:
    rclpy.spin_once(n,timeout_sec=.1)
    if len(got)==10 and b.can_transform('panda_link0','robotiq_pinch',rclpy.time.Time()): break
stamp=datetime.datetime.now(datetime.timezone.utc).strftime('%Y%m%dT%H%M%S.%fZ'); out=pathlib.Path('observations')/stamp;out.mkdir(parents=True)
meta={'time_utc':stamp,'joints':dict(zip(got['joints'].name,got['joints'].position)),'tf':{},'cameras':{}}
frames=['panda_link8','robotiq_pinch']
for c in m['sensors']:
    if c['kind']!='camera':continue
    name=c['name']; info=got[name+'_info']; frames.append(info.header.frame_id)
    cv2.imwrite(str(out/(name+'.png')),CvBridge().imgmsg_to_cv2(got[name+'_color'],'bgr8'))
    np.save(out/(name+'_depth.npy'),CvBridge().imgmsg_to_cv2(got[name+'_depth'],'passthrough'))
    meta['cameras'][name]={'k':list(info.k),'frame':info.header.frame_id,'encoding':got[name+'_depth'].encoding}
for f in frames:
    try:
        t=b.lookup_transform('panda_link0',f,rclpy.time.Time()).transform
        meta['tf'][f]={'p':[t.translation.x,t.translation.y,t.translation.z],'q':[t.rotation.x,t.rotation.y,t.rotation.z,t.rotation.w]}
    except Exception as e:meta['tf'][f]={'error':str(e)}
(out/'state.json').write_text(json.dumps(meta,indent=2));pathlib.Path('observations/latest.txt').write_text(str(out)); print(str(out));print(json.dumps(meta,indent=2));n.destroy_node();rclpy.shutdown()
