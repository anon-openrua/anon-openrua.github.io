#!/usr/bin/env python3
"""Save fresh wrist/fixed RGB-D, joints and TF to a UTC-stamped directory.

Dependencies: ROS 2 rclpy/sensor_msgs/tf2_ros/cv_bridge, numpy, scipy, OpenCV,
PyYAML; supplied by the robot runtime. Run only while the arm is stationary.
Usage: python3 scripts/observe.py LABEL
"""
import json
import sys
import time
from datetime import datetime, timezone
from pathlib import Path
import cv2
import numpy as np
import rclpy
import yaml
from cv_bridge import CvBridge
from sensor_msgs.msg import Image, CameraInfo, JointState
from tf2_ros import Buffer, TransformListener
from scipy.spatial.transform import Rotation

rclpy.init()
node = rclpy.create_node('task_observe')
buf = Buffer()
listener = TransformListener(buf, node)
manifest = yaml.safe_load(Path('/robot/machine.yaml').read_text())
got = {}
subs = []
for cam in manifest['sensors']:
    if cam['kind'] != 'camera':
        continue
    for stream, cls in [('color', Image), ('depth', Image), ('info', CameraInfo)]:
        key = cam['name'] + '_' + stream
        subs.append(node.create_subscription(cls, cam[stream], lambda m, k=key: got.setdefault(k, m), 1))
subs.append(node.create_subscription(JointState, '/joint_states', lambda m: got.setdefault('joints', m), 1))
deadline = time.monotonic() + 12
while time.monotonic() < deadline:
    rclpy.spin_once(node, timeout_sec=0.1)
    if len(got) == 7 and buf.can_transform('panda_link0', 'zed_wrist_left_camera_optical_frame', rclpy.time.Time()):
        break
assert len(got) == 7, list(got)
out = Path('/workspace/observations') / (datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%SZ') + '_' + sys.argv[1])
out.mkdir(parents=True)
meta = {'joints': dict(zip(got['joints'].name, got['joints'].position)), 'transforms': {}}
for name in ['d405', 'zed_wrist']:
    color, depth, info = [got[name + '_' + s] for s in ['color', 'depth', 'info']]
    cv2.imwrite(str(out / (name + '.png')), CvBridge().imgmsg_to_cv2(color, 'bgr8'))
    a = CvBridge().imgmsg_to_cv2(depth, 'passthrough').astype(float)
    if depth.encoding == '16UC1':
        a *= 0.001
    np.save(out / (name + '_depth_m.npy'), a)
    meta[name] = {'K': list(info.k), 'frame': info.header.frame_id, 'stamp': [color.header.stamp.sec, color.header.stamp.nanosec]}
for frame in ['panda_link8', 'robotiq_pinch', 'zed_wrist_left_camera_optical_frame']:
    t = buf.lookup_transform('panda_link0', frame, rclpy.time.Time()).transform
    q = [t.rotation.x, t.rotation.y, t.rotation.z, t.rotation.w]
    T = np.eye(4)
    T[:3,:3] = Rotation.from_quat(q).as_matrix()
    T[:3,3] = [t.translation.x, t.translation.y, t.translation.z]
    meta['transforms'][frame] = {'matrix': T.tolist(), 'quaternion': q}
(out / 'state.json').write_text(json.dumps(meta, indent=2))
print(out)
print(json.dumps(meta['transforms']))
node.destroy_node()
rclpy.shutdown()
