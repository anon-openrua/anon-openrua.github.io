#!/usr/bin/env python3
"""Save fresh RGB/depth, intrinsics, joints and TF; no robot motion.
Requires machine ROS environment, rclpy, cv_bridge, numpy, scipy, PyYAML.
Usage: python3 scripts/observe.py LABEL
"""
import json, sys, time
from pathlib import Path
from datetime import datetime, timezone
import cv2, numpy as np, yaml, rclpy
from cv_bridge import CvBridge
from sensor_msgs.msg import Image, CameraInfo, JointState
from tf2_ros import Buffer, TransformListener
from scipy.spatial.transform import Rotation
rclpy.init(); node=rclpy.create_node('project_observe')
buf=Buffer(); listener=TransformListener(buf,node); bridge=CvBridge(); got={}
m=yaml.safe_load(Path('/robot/machine.yaml').read_text())
folder=Path('observations')/(datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%SZ')+'_'+sys.argv[1]); folder.mkdir(parents=True)
cams=[s for s in m['sensors'] if s['kind']=='camera']
subs=[]
for c in cams:
    for k,typ in [('color',Image),('depth',Image),('info',CameraInfo)]:
        key=c['name']+'_'+k
        subs.append(node.create_subscription(typ,c[k],lambda msg,key=key:got.setdefault(key,msg),1))
subs.append(node.create_subscription(JointState,'/joint_states',lambda msg:got.setdefault('joints',msg),1))
end=time.monotonic()+15
while len(got)<7 and time.monotonic()<end: rclpy.spin_once(node,timeout_sec=.1)
for _ in range(10): rclpy.spin_once(node,timeout_sec=.05)
out={'acquired_utc':datetime.now(timezone.utc).isoformat(),'frames':{}}
for c in cams:
    name=c['name']; rgb=got[name+'_color']; dep=got[name+'_depth']; info=got[name+'_info']
    cv2.imwrite(str(folder/(name+'.png')),bridge.imgmsg_to_cv2(rgb,'bgr8'))
    np.save(folder/(name+'_depth.npy'),bridge.imgmsg_to_cv2(dep,'passthrough'))
    out[name]={'k':list(info.k),'frame':info.header.frame_id,'depth_encoding':dep.encoding,'color_stamp':[rgb.header.stamp.sec,rgb.header.stamp.nanosec],'depth_stamp':[dep.header.stamp.sec,dep.header.stamp.nanosec]}
for frame in ['panda_link8','robotiq_pinch']+[got[c['name']+'_info'].header.frame_id for c in cams]:
    try:
        t=buf.lookup_transform('panda_link0',frame,rclpy.time.Time()).transform
        q=[t.rotation.x,t.rotation.y,t.rotation.z,t.rotation.w]; p=[t.translation.x,t.translation.y,t.translation.z]
        mat=np.eye(4); mat[:3,:3]=Rotation.from_quat(q).as_matrix(); mat[:3,3]=p
        out['frames'][frame]={'xyz':p,'quat_xyzw':q,'matrix':mat.tolist()}
    except Exception as e: out['frames'][frame]={'error':str(e)}
js=got['joints']; out['joints']=dict(zip(js.name,js.position))
(folder/'state.json').write_text(json.dumps(out,indent=2)); print(folder); print(json.dumps(out['frames'],indent=2)); rclpy.shutdown()
