#!/usr/bin/env python3
"""Save fresh RGB/depth, camera intrinsics, joints and TF in a UTC directory.
Requires the ROS environment plus rclpy, tf2_ros, cv_bridge, numpy, OpenCV,
scipy and PyYAML supplied by /robot. Does not command motion.
"""
import json, time
from pathlib import Path
from datetime import datetime, timezone
import cv2, numpy as np, yaml, rclpy
from cv_bridge import CvBridge
from sensor_msgs.msg import Image, CameraInfo, JointState
from tf2_ros import Buffer, TransformListener
from scipy.spatial.transform import Rotation

rclpy.init()
n = rclpy.create_node('project_observe')
b = Buffer(); listener = TransformListener(b, n)
m = yaml.safe_load(Path('/robot/machine.yaml').read_text())
cams = [x for x in m['sensors'] if x['kind'] == 'camera']
got = {}; subs = []
for c in cams:
    for k, typ in [('color', Image), ('depth', Image), ('info', CameraInfo)]:
        key = c['name'] + '_' + k
        subs.append(n.create_subscription(typ, c[k], lambda msg, key=key: got.update({key: msg}), 1))
subs.append(n.create_subscription(JointState, '/joint_states', lambda msg: got.update(joints=msg), 1))
end = time.monotonic() + 12
while time.monotonic() < end:
    rclpy.spin_once(n, timeout_sec=.1)
    if len(got) == 7 and b.can_transform('panda_link0', 'panda_link8', rclpy.time.Time()):
        break
out = Path('/workspace/observations') / datetime.now(timezone.utc).strftime('%Y%m%dT%H%M%SZ')
out.mkdir(parents=True)
data = {'acquired_utc': datetime.now(timezone.utc).isoformat(), 'tf': {}}
for c in cams:
    name = c['name']; info = got[name + '_info']
    cv2.imwrite(str(out / (name + '.png')), CvBridge().imgmsg_to_cv2(got[name + '_color'], 'bgr8'))
    np.save(out / (name + '_depth.npy'), CvBridge().imgmsg_to_cv2(got[name + '_depth'], 'passthrough'))
    data[name] = {'k': list(info.k), 'frame': info.header.frame_id, 'depth_encoding': got[name + '_depth'].encoding}
for frame in ['panda_link8', 'robotiq_pinch'] + [got[c['name'] + '_info'].header.frame_id for c in cams]:
    try:
        t = b.lookup_transform('panda_link0', frame, rclpy.time.Time()).transform
        p = [t.translation.x, t.translation.y, t.translation.z]
        q = [t.rotation.x, t.rotation.y, t.rotation.z, t.rotation.w]
        data['tf'][frame] = {'p': p, 'q': q}
    except Exception as e:
        data['tf'][frame] = {'error': str(e)}
j = got['joints']; data['joints'] = dict(zip(j.name, j.position))
(out / 'state.json').write_text(json.dumps(data, indent=2))
print(out)
print(json.dumps(data['tf'], indent=2))
n.destroy_node(); rclpy.shutdown()
