from datetime import datetime, timezone

entries=[]

    dest=p.with_name(stamp.strftime('%Y%m%dT%H%M%S.%fZ_')+p.name)
    p.rename(dest)
    entries.append({'file':dest.name,'captured_utc_approx':stamp.isoformat(),'original_label':p.name})
(root/'index.json').write_text(json.dumps(entries,indent=2)+'\n')
print(json.dumps([e for e in entries if e['original_label'].startswith('final_')],indent=2))
